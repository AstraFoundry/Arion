"""Validate complete reviewed evidence and safe standalone HTML embedding."""
import json
import unittest
import sys
import types
import report
class ReportTests(unittest.TestCase):
    @classmethod
    def setUpClass(cls):
        cls.data=report.collect()
    def test_complete_evidence(self):
        data=self.data
        self.assertEqual(len(data['cases']),4263)
        self.assertEqual(data['naturalAsrPreparation']['sources'],120)
        self.assertFalse(data['naturalAsrPreparation']['newInference'])
        self.assertEqual(data['timingIntegrity'],dict(observations=396,flagged=9))
        self.assertEqual(len(data['rounds']),44)
        self.assertEqual(sum(c['actualMessages'] is not None for c in data['cases']),2779)
        self.assertEqual(sum(c['modelSource'] is not None for c in data['cases']),2659)
        self.assertEqual(sum(bool(c['reviewQualifier']) for c in data['cases']),1)
        self.assertEqual(data['naturalTextReview']['reviewed'],120)
        self.assertEqual(len(data['audits']),113)
        self.assertEqual(len(data['ocrFixtures']),12)
        self.assertEqual(sum(len(f['observations']) for f in data['ocrFixtures']),24)
        self.assertEqual(sum(r['count'] for r in data['rounds']),4463)
        keys=[(c['run'],c['variant'],c['id']) for c in data['cases']]
        self.assertEqual(len(keys),len(set(keys)))
        self.assertTrue(all(c['scene'] and c['effectiveScene'] and c['template'] for c in data['cases']))
        self.assertTrue(all('未测自动命中' in c['assignment'] for c in data['cases']))
    def test_uncertain_meaning_and_new_natural_inference(self):
        data=self.data;self.assertEqual(sum(c['faithful'] is None for c in data['cases']),12)
        self.assertEqual(data['naturalCandidateReview']['generations'],1080)
        self.assertEqual(data['cleanupCopula']['version'],'speech-spans-v2')
        self.assertEqual(data['priming']['warmupGenerations'],32)
        self.assertEqual(data['priming']['crossOrderRawParity'],174)
        self.assertEqual(data['workScope']['v4']['groups']['candidate/historical-synthetic-regression']['faithful'],94)
        self.assertEqual(data['workScope']['v4']['groups']['candidate/historical-synthetic-regression']['useful'],87)
        self.assertEqual(data['prefixCache']['rawParity'],174)
        self.assertEqual(data['prefixCache']['cacheHits'],164)
        self.assertEqual(data['continuousChain']['selectorCalls'],18)
        self.assertEqual(data['continuousChain']['developmentUseful'],36)
        self.assertTrue(all(not c['useful'] for c in data['cases'] if c['faithful'] is None))
    def test_user_aligned_view_preserves_original(self):
        data=self.data;self.assertEqual(data['spellingAlignment']['reclassified'],16)
        self.assertEqual(sum(c.get('alignedReview',{}).get('faithful',False) for c in data['cases']),16)
        self.assertTrue(all(c['faithful'] is False for c in data['cases'] if c.get('alignedReview')))
        self.assertEqual(sum(c.get('stage')=='selection' for c in data['cases']),216)
        self.assertTrue(all(c['delivery'] is None for c in data['cases'] if c.get('stage')=='selection'))
    def test_report_loaders_preserve_module_and_path(self):
        sentinel=types.ModuleType('run');previous=sys.modules.get('run');before=list(sys.path)
        sys.modules['run']=sentinel
        try:
            self.assertEqual(len(report.collect()['cases']),4263)
            self.assertIs(sys.modules['run'],sentinel)
            self.assertEqual(sys.path,before)
        finally:
            sys.modules.pop('run',None)
            if previous is not None:sys.modules['run']=previous
    def test_hostile_text_remains_data(self):
        attack={'text':'</script><script>alert("wrong")</script>&OCR'}
        encoded=report.encode_data(attack)
        self.assertNotIn('</script>',encoded)
        self.assertNotIn('<',encoded)
        self.assertEqual(json.loads(encoded),attack)
    def test_strict_delivery_origin(self):
        data=self.data
        for item in data['cases']:
            if item['delivery']:
                accepted=item['strict']=='未启用' or (item['strict']=='accepted' and item['envelope']=='accepted')
                self.assertEqual(item['delivery']['text'],item['output'] if accepted else item['cleanup'])
if __name__=='__main__':unittest.main()
