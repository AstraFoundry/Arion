"""Compare full original and replacement sentences with counterbalanced option labels."""
import importlib.util
import json
import pathlib

HERE = pathlib.Path(__file__).resolve().parent
ARMS = ['baseline', 'balanced', 'sentence-ab', 'sentence-ba']
SENTENCE_ARMS = {'sentence-ab', 'sentence-ba'}


def module(name, path):
    spec = importlib.util.spec_from_file_location(name, path)
    value = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(value)
    return value


prior = module('sentence_choice_prior', HERE.parent/'rewrite-selector-screen-v1/pipeline.py')


def original_label(arm):
    if arm not in SENTENCE_ARMS:
        raise ValueError('Not a sentence choice arm')
    return 'A' if arm == 'sentence-ab' else 'B'


def decision_choices(arm, selection, prepared):
    if arm not in ARMS:
        raise ValueError('Unknown sentence choice arm')
    if arm not in SENTENCE_ARMS:
        return selection
    if not isinstance(selection, dict) or set(selection) != {h['id'] for h in prepared['hypotheses']}:
        raise ValueError('Missing or fabricated sentence choice id')
    if any(value not in ['A', 'B'] for value in selection.values()):
        raise ValueError('Invalid sentence choice label')
    return {key: 'keep' if value == original_label(arm) else 'replace' for key, value in selection.items()}


class SentenceCandidates:
    def __init__(self, arm):
        self.arm = arm

    def apply(self, raw, prepared):
        choices = decision_choices(self.arm, json.loads(raw), prepared)
        return prior.candidates_for('baseline').apply(json.dumps(choices), prepared)


def candidates_for(arm):
    if arm not in ARMS:
        raise ValueError('Unknown sentence choice arm')
    return SentenceCandidates(arm) if arm in SENTENCE_ARMS else prior.candidates_for(arm)


def transform_data(data, arm):
    original = original_label(arm)
    for hypothesis in data['hypotheses']:
        comparisons = hypothesis.pop('candidateComparisons')
        if not comparisons or any(item['originalSource'] != data['source'] for item in comparisons):
            raise ValueError('Changed comparison source')
        replacement = [item['replacementSource'] for item in comparisons]
        hypothesis['sentenceChoices'] = {'A': [data['source']] if original == 'A' else replacement,
                                         'B': [data['source']] if original == 'B' else replacement}
    return data


def selection_payload(arm, sample, cleanup):
    candidates_for(arm)
    request, prepared = prior.selection_payload('baseline' if arm == 'baseline' else 'balanced', sample, cleanup)
    if arm not in SENTENCE_ARMS:
        return request, prepared
    prompt = json.loads((HERE/'prompts.json').read_text())
    request['messages'][0]['content'] = prompt['selectionSystem']
    for message in request['messages']:
        if message['role'] == 'user':
            data = transform_data(json.loads(message['content']), arm)
            message['content'] = json.dumps(data, ensure_ascii=False, separators=(',', ':'))
        elif message['role'] == 'assistant':
            choices = json.loads(message['content'])
            record = {key: original_label(arm) if value == 'keep' else ('B' if original_label(arm) == 'A' else 'A') for key, value in choices.items()}
            message['content'] = json.dumps(record, ensure_ascii=False, separators=(',', ':'))
    for field in request['response_format']['schema']['properties'].values():
        field['enum'] = ['A', 'B']
    return request, prepared


def template_identity(arm, stage, scene):
    candidates_for(arm)
    if stage != 'selection':
        raise ValueError('Sentence choice study has no formatter')
    if arm not in SENTENCE_ARMS:
        return prior.template_identity(arm, stage, scene)
    return dict(version='selector-sentence-choice-v1-'+arm, path='benchmarks/reproduction/rewrite-selector-sentence-choice-v1/prompts.json')


def baseline_projection(request, arm):
    candidates_for(arm)
    result = json.loads(json.dumps(request, ensure_ascii=False))
    if arm in SENTENCE_ARMS:
        result['messages'][0]['content'] = json.loads((HERE.parent/'rewrite-selector-screen-v1/templates/baseline/prompts.json').read_text())['selectionSystem']
        original = original_label(arm)
        other = 'B' if original == 'A' else 'A'
        for message in result['messages']:
            if message['role'] == 'user':
                data = json.loads(message['content'])
                for hypothesis in data['hypotheses']:
                    options = hypothesis.pop('sentenceChoices')
                    if set(options) != {'A', 'B'} or options[original] != [data['source']]:
                        raise ValueError('Changed original sentence option')
                    hypothesis['candidateComparisons'] = [dict(originalSource=data['source'], replacementSource=value) for value in options[other]]
                message['content'] = json.dumps(data, ensure_ascii=False, separators=(',', ':'))
            elif message['role'] == 'assistant':
                choices = json.loads(message['content'])
                message['content'] = json.dumps({key: 'keep' if value == original else 'replace' for key, value in choices.items()}, separators=(',', ':'))
        for field in result['response_format']['schema']['properties'].values():
            field['enum'] = ['keep', 'replace']
    return prior.baseline_projection(result, 'baseline' if arm == 'baseline' else 'balanced')
