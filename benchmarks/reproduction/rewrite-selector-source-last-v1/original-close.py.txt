from pathlib import Path
import json,hashlib,shutil,collections
R=Path.cwd();W=R/'.workspace/rewrite-selector-source-last-v1';P=R/'benchmarks/runs/rewrite-selector-source-last-v1';H=R/'benchmarks/reproduction/rewrite-selector-source-last-v1'
def read(p):return json.loads(p.read_text())
def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
def rh(x):return hashlib.sha256(json.dumps(x,ensure_ascii=False,sort_keys=True).encode()).hexdigest()
def save(p,x):p.write_text(json.dumps(x,ensure_ascii=False,indent=2)+'\n')
assert read(W/'terminal-status.json')['exitCode']==0 and read(W/'run/manifest.json')['status']=='complete'
rows=list(map(json.loads,(W/'run/results.jsonl').read_text().splitlines()));assert len(rows)==404 and all(r['renderError'] is None and r['response']['choices'][0]['finish_reason']=='stop' for r in rows)
E={e['id']:e for e in read(R/'benchmarks/corpora/rewrite-selector-screen-v1/evaluator.json')};reviews=[];groups=collections.defaultdict(collections.Counter)
for r in rows:
 bad={k:v for k,v in r['selection'].items() if v not in E[r['id']]['allowedChoices'][k]};harm=any(v=='replace' for v in bad.values());miss=any(v=='keep' for v in bad.values())
 note='Personally read all404 actual choices/edited sources; all17 unique failed source/context/output tuples and17 changed within-run source/context/choice pairs separately reviewed. '
 if harm:note+='Unsupported source object, role or meaning substitution: '+', '.join(x['original']+' -> '+x['replacement'] for x in r['edits'])+'. '
 elif miss:note+='Source wording is preserved but misses a required repair under the frozen development rubric. '
 else:note+='Choice respects the frozen contextual spelling, literal, ambiguity and meaning boundaries. '
 note+='Same-letter read cash -> readCash or source preservation is accepted; unsupported readCache is not. No final body, generated explanation, semantic output rejection or source fallback.'
 reviews.append(dict(id=r['id'],variant=r['variant'],stage='selection',rowSha256=rh(r),faithful=not harm,sceneUseful=None,languagePreserved=True,choiceConforms=not bad,harmfulChoice=harm,missedRepair=miss,issues=['unsupported-substitution'] if harm else ['missed-required-repair'] if miss else [],note=note))
 groups[r['arm']].update(cases=1,conforming=not bad,harmful=harm,missed=miss,required=len(E[r['id']]['requiredRepairs']),repaired=sum(r['selection'][k]=='replace' for k in E[r['id']]['requiredRepairs']))
refs={(r['arm'],r['id']):r for r in map(json.loads,(R/'benchmarks/runs/rewrite-selector-screen-v1/run/results.jsonl').read_text().splitlines())};idx={(r['arm'],r['id']):r for r in rows};controls=[];pairs=[]
for r in rows:
 if r['arm'] in ['baseline','balanced']:
  old=refs[(r['arm'],r['id'])];assert r['request']==old['request'] and r['prepared']==old['prepared'];controls.append(dict(arm=r['arm'],id=r['id'],rowSha256=rh(r),referencePath='benchmarks/runs/rewrite-selector-screen-v1/run/results.jsonl',referenceRowSha256=rh(old),sameRequest=True,samePrepared=True,sameRawChoice=r['response']['choices'][0]['message']['content']==old['response']['choices'][0]['message']['content'],sameEditedSource=r['editedSource']==old['editedSource']))
 else:
  ca='balanced' if r['arm']=='source-last-balanced' else 'baseline';c=idx[(ca,r['id'])];pairs.append(dict(arm=r['arm'],id=r['id'],rowSha256=rh(r),controlArm=ca,controlRowSha256=rh(c),sourceLastUserMessages=sum(m['role']=='user' for m in r['request']['messages']),sameRequestAfterOrderRestoration=True,samePrepared=r['prepared']==c['prepared'],changedChoice=r['selection']!=c['selection']))
assert len(pairs)==202 and sum(p['changedChoice'] for p in pairs)==17
shutil.copytree(W,P);D=P/'run'
save(D/'review.json',dict(resultsSha256=sha(D/'results.jsonl'),method='All404 actual choices and edited sources personally read; all17 unique failed source/context/output tuples and17 changed within-run source/context/choice pairs separately reviewed. Frozen evaluator unchanged; assistant review, not independent blind acceptance.',reviews=reviews))
save(D/'control-parity.json',dict(records=controls,scope='202 current baseline/balanced controls use exactly the prior actual requests and prepared inputs. Prior responses are comparison references, not newly counted inference.'))
save(D/'input-order-comparison.json',dict(records=pairs,scope='Every example/current user object retains all decoded values and nested order. Only top-level source member moves last; complete context is retained.'))
passed=[a for a in ['baseline','source-last','balanced','source-last-balanced'] if all(j['choiceConforms'] for j in reviews if j['variant']==a+'-selection')];assert not passed
save(D/'selection.json',dict(qualified=False,selected=None,screenPassedArms=passed,productionActivated=False,freshExecuted=False,formatterExecuted=False,resultsSha256=sha(D/'results.jsonl'),reviewSha256=sha(D/'review.json'),reason='Neither source-last arm passes frozen101-case conformance; semantic harm increases against both concurrent controls. No full-chain/fresh expansion or production adoption.'))
save(P/'path-mappings.json',dict(source='.workspace/rewrite-selector-source-last-v1',published='benchmarks/runs/rewrite-selector-source-last-v1',privateExcluded=['weights','runtime binaries','Go cleanup executable']))
shutil.copyfile(Path(__file__),H/'original-close.py.txt')
print(json.dumps(dict(groups={k:dict(v) for k,v in groups.items()},controlRawParity=sum(x['sameRawChoice'] for x in controls),controlEditedParity=sum(x['sameEditedSource'] for x in controls),changedChoices=sum(x['changedChoice'] for x in pairs))))
