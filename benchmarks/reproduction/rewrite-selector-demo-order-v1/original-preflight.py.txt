"""Verify all reordered prompts fit the native context without generating answers."""
from pathlib import Path
import sys,json,hashlib
R=Path.cwd();H=R/'benchmarks/reproduction/rewrite-selector-demo-order-v1';W=R/'.workspace/rewrite-selector-demo-order-v1'
sys.path[:0]=[str(H),str(H.parent)]
import pipeline
import gguf_rewrite as engine
from gguf_session_v2 import Session
f=engine.read(R/'benchmarks/reproduction/rewrite-selector-screen-v1/freeze.json');engine.MODEL=R/f['modelPath'];engine.verify(f)
D=W/'preflight';D.mkdir(exist_ok=False);engine.write(D/'status.json',dict(status='running',modelGenerations=0))
preps=list(map(json.loads,(R/'benchmarks/runs/rewrite-selector-screen-v1/run/preparations.jsonl').read_text().splitlines()));references={(r['arm'],r['id']):r for r in map(json.loads,(R/'benchmarks/runs/rewrite-selector-screen-v1/run/results.jsonl').read_text().splitlines())};worker=None;count=0;max_tokens=0

def rh(value):return hashlib.sha256(json.dumps(value,ensure_ascii=False,sort_keys=True).encode()).hexdigest()

try:
 worker=Session(D,'token-budget',18823,gpu_layers=99,verbosity=5)
 with (D/'requests.jsonl').open('x') as stream:
  for arm in pipeline.ARMS:
   for p in preps:
    if not p['prepared']['hypotheses']:continue
    request,prepared=pipeline.selection_payload(arm,p['sample'],p['cleanupObservation']['cleanup'])
    assert prepared==p['prepared']
    applied=engine.base.request(18823,'/apply-template',request,timeout=30)
    token_request=dict(content=applied['prompt'],add_special=False,parse_special=True,with_pieces=False)
    tokenized=engine.base.request(18823,'/tokenize',token_request,timeout=30)
    ids=tokenized['tokens'];assert all(isinstance(i,int) for i in ids)
    if len(ids)+request['max_tokens']>4096:raise ValueError('Complete prompt and unchanged output reserve exceed native context')
    reference=None
    if arm in ['baseline','balanced']:
     prior=references[(arm,p['sample']['id'])]
     assert request==prior['request']
     if len(ids)!=prior['response']['usage']['prompt_tokens']:raise ValueError('Native template/token count differs from prior actual control request')
     reference=dict(path='benchmarks/runs/rewrite-selector-screen-v1/run/results.jsonl',rowSha256=rh(prior),actualPromptTokens=prior['response']['usage']['prompt_tokens'])
    record=dict(arm=arm,id=p['sample']['id'],request=request,requestSha256=rh(request),sourcePreparationReference=dict(path='benchmarks/runs/rewrite-selector-screen-v1/run/preparations.jsonl',rowSha256=rh(p)),applyTemplateResponse=applied,tokenizeRequest=dict(add_special=False,parse_special=True,with_pieces=False),tokenizeResponse=tokenized,promptTokens=len(ids),outputReserve=request['max_tokens'],contextLimit=4096,controlReference=reference,modelGenerations=0)
    stream.write(json.dumps(record,ensure_ascii=False)+'\n');stream.flush();count+=1;max_tokens=max(max_tokens,len(ids))
 command=worker.command;worker.close();worker=None
 assert count==404
 engine.write(D/'status.json',dict(status='complete',modelGenerations=0,templateRequests=count,tokenizationRequests=count,maximumPromptTokens=max_tokens,outputReserve=128,contextLimit=4096,requestsSha256=engine.sha(D/'requests.jsonl'),model=f['model'],command=command,meaning='Native template/tokenizer compatibility and context budget only. No model answer or real ASR/OCR/Go preparation is generated. Controls match prior actual prompt usage exactly.'))
 print('Native token-budget preflight complete',count,'requests, max prompt',max_tokens,'reserve128, generations0',flush=True)
except Exception as error:
 if worker:worker.abort()
 engine.write(D/'status.json',dict(status='failed',modelGenerations=0,completedRequests=count,maximumPromptTokens=max_tokens,error=repr(error)));raise
