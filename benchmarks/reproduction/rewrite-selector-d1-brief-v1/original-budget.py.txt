"""Measure all proposed native question/option budgets without model inference."""
from pathlib import Path
import hashlib,importlib.util,json,socket,subprocess,time,urllib.request,urllib.error
R=Path.cwd();W=Path(__file__).resolve().parent;F=json.loads((W/'freeze.json').read_text());OUT=W/'budget'
def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
def save(p,x):p.write_text(json.dumps(x,ensure_ascii=False,indent=2)+'\n')
for name,expected in F['files'].items():
 if sha(R/name)!=expected:raise ValueError('Changed proposed budget source')
rt=R/F['runtimePath'];model=R/F['modelPath']
for name,expected in F['runtimeFiles'].items():
 if sha(rt/name)!=expected:raise ValueError('Changed actual tokenizer runtime')
if sha(model)!=F['modelSha256']:raise ValueError('Changed actual model tokenizer')
spec=importlib.util.spec_from_file_location('existing_d1_native_renderer',R/'benchmarks/reproduction/rewrite-selector-d1-policy-v1/budget.py');helper=importlib.util.module_from_spec(spec);spec.loader.exec_module(helper)
records=json.loads((W/'payloads.json').read_text())['records'];OUT.mkdir(exist_ok=False)
with socket.socket() as sock:sock.bind(('127.0.0.1',0));port=sock.getsockname()[1]
command=[str(rt/'llama-server'),'-m',str(model),'-ngl','99','-t','4','-c','4096','-np','1','-b','4096','-ub','4096','--no-warmup','-lv','5','--host','127.0.0.1','--port',str(port)];save(OUT/'command.json',command);cache={};results=[]
def tokenize(text):
 if text in cache:return cache[text]
 payload=dict(content=text,add_special=False,parse_special=True,with_pieces=False);request=urllib.request.Request(f'http://127.0.0.1:{port}/tokenize',data=json.dumps(payload,ensure_ascii=False).encode(),headers={'Content-Type':'application/json'},method='POST')
 with urllib.request.urlopen(request,timeout=20) as response:raw=response.read();data=json.loads(raw)
 cache[text]=data['tokens']
 with (OUT/'native-calls.jsonl').open('a') as trace:trace.write(json.dumps(dict(endpoint='/tokenize',request=payload,response=data,rawResponse=raw.decode()),ensure_ascii=False)+'\n')
 return cache[text]
with (OUT/'server.log').open('wb') as log:
 p=subprocess.Popen(command,stdout=log,stderr=subprocess.STDOUT);start=time.monotonic()
 try:
  ready=False
  while p.poll() is None and time.monotonic()-start<45:
   try:
    with urllib.request.urlopen(f'http://127.0.0.1:{port}/health',timeout=1) as response:ready=response.status==200
    if ready:break
   except (urllib.error.URLError,TimeoutError):pass
   time.sleep(.2)
  if not ready:raise RuntimeError('Actual tokenizer did not start')
  with urllib.request.urlopen(f'http://127.0.0.1:{port}/props',timeout=5) as response:(OUT/'props.json').write_bytes(response.read())
  for record in records:
   for identity,question in record['request']['questions'].items():
    request=dict(state=record['request']['state'],questions={identity:question});parts=helper.render(request).split(helper.SEP);pieces=[]
    for text in parts[1:]:
     kind='other';limit=None
     for marker,label,n in [(helper.STATE,'state',None),(helper.QUESTION,'question',96),(helper.OPTION,'option',45)]:
      if text.startswith(marker):text=text[len(marker):];kind=label;limit=n;break
     ids=tokenize(text);pieces.append(dict(kind=kind,text=text,tokens=ids,limit=limit,overBudget=limit is not None and len(ids)>limit))
    count=sum(len(x['tokens']) for x in pieces);item=dict(variant=record['variant'],id=record['id'],questionId=identity,sourceReference=record['sourceReference'],request=request,pieces=pieces,unclippedInputTokens=count,passesBudget=not any(x['overBudget'] for x in pieces) and count<=4096,modelResponses=0)
    results.append(item)
    with (OUT/'requests.jsonl').open('a') as trace:trace.write(json.dumps(item,ensure_ascii=False)+'\n')
 finally:
  if p.poll() is None:
   p.terminate()
   try:p.wait(timeout=10)
   except subprocess.TimeoutExpired:p.kill();p.wait(timeout=5)
groups={}
for variant in ['sentence','spelling']:
 rows=[r for r in results if r['variant']==variant];groups[variant]=dict(sourceCasesProposed=101,inferredSourceCasesProposed=89,questionsMeasured=len(rows),forcedOriginalCases=12,passes=sum(r['passesBudget'] for r in rows),maximumQuestionTokens=max(len(x['tokens']) for r in rows for x in r['pieces'] if x['kind']=='question'),maximumOptionTokens=max(len(x['tokens']) for r in rows for x in r['pieces'] if x['kind']=='option'),maximumInputTokens=max(r['unclippedInputTokens'] for r in rows),failures=[dict(id=r['id'],questionId=r['questionId']) for r in rows if not r['passesBudget']])
summary=dict(status='complete',modelResponses=0,generatedTokens=0,nativeTokenizationCalls=len(cache),groups=groups,requestsSha256=sha(OUT/'requests.jsonl'),nativeExitCode=p.returncode,limits='Proposed input budget only, not quality or native typed responses.12 forced-original cases per variant are not model decisions. Complete current source/OCR retained, no demonstration/policy transcript. No truncation allowed for future inference.')
save(OUT/'summary.json',summary);print(json.dumps(summary))
