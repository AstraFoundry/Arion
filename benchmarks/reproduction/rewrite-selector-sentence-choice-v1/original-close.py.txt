from pathlib import Path
import json,hashlib,shutil,collections
R=Path.cwd();W=R/'.workspace/rewrite-selector-sentence-choice-v1';P=R/'benchmarks/runs/rewrite-selector-sentence-choice-v1';H=R/'benchmarks/reproduction/rewrite-selector-sentence-choice-v1'
def read(p):return json.loads(p.read_text())
def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
def rh(x):return hashlib.sha256(json.dumps(x,ensure_ascii=False,sort_keys=True).encode()).hexdigest()
def save(p,x):p.write_text(json.dumps(x,ensure_ascii=False,indent=2)+'\n')
assert read(W/'terminal-status.json')['exitCode']==0 and read(W/'run/manifest.json')['status']=='complete'
rows=list(map(json.loads,(W/'run/results.jsonl').read_text().splitlines()));assert len(rows)==404 and all(r['renderError'] is None and r['response']['choices'][0]['finish_reason']=='stop' for r in rows)
E={e['id']:e for e in read(R/'benchmarks/corpora/rewrite-selector-screen-v1/evaluator.json')};reviews=[];groups=collections.defaultdict(collections.Counter)
def choices(r):return {k:('keep' if v==('A' if r['arm']=='sentence-ab' else 'B') else 'replace') for k,v in r['selection'].items()} if r['arm'].startswith('sentence-') else r['selection']
for r in rows:
 c=choices(r);bad={k:v for k,v in c.items() if v not in E[r['id']]['allowedChoices'][k]};harm=any(v=='replace' for v in bad.values());miss=any(v=='keep' for v in bad.values())
 note='Personally read all404 actual raw choices and edited sources. All202 A/B cases read with complete source/context in explicit ranges; all28 unique failed source/context/output tuples and32 changed balanced-control pairs separately reviewed. '
 if harm:note+='Unsupported source object, role or meaning substitution: '+', '.join(x['original']+' -> '+x['replacement'] for x in r['edits'])+'. '
 elif miss:note+='Source wording is preserved but misses required contextual spelling repair under the frozen development rubric. '
 else:note+='Normalized choice respects the frozen spelling/literal/ambiguity/semantic boundary. '
 note+='Raw A/B and normalized keep/replace remain separate; neither is a final body. Same-letter canonicalization or preservation is allowed, unsupported readCache is not. No semantic output rejection/source fallback.'
 reviews.append(dict(id=r['id'],variant=r['variant'],stage='selection',rowSha256=rh(r),faithful=not harm,sceneUseful=None,languagePreserved=True,choiceConforms=not bad,harmfulChoice=harm,missedRepair=miss,normalizedChoices=c,issues=['unsupported-substitution'] if harm else ['missed-required-repair'] if miss else [],note=note))
 groups[r['arm']].update(cases=1,conforming=not bad,harmful=harm,missed=miss,required=len(E[r['id']]['requiredRepairs']),repaired=sum(c[k]=='replace' for k in E[r['id']]['requiredRepairs']))
refs={(r['arm'],r['id']):r for r in map(json.loads,(R/'benchmarks/runs/rewrite-selector-screen-v1/run/results.jsonl').read_text().splitlines())};idx={(r['arm'],r['id']):r for r in rows};controls=[];pairs=[]
for r in rows:
 if r['arm'] in ['baseline','balanced']:
  old=refs[(r['arm'],r['id'])];assert r['request']==old['request'] and r['prepared']==old['prepared'];controls.append(dict(arm=r['arm'],id=r['id'],rowSha256=rh(r),referencePath='benchmarks/runs/rewrite-selector-screen-v1/run/results.jsonl',referenceRowSha256=rh(old),sameRequest=True,samePrepared=True,sameRawChoice=r['response']['choices'][0]['message']['content']==old['response']['choices'][0]['message']['content'],sameEditedSource=r['editedSource']==old['editedSource']))
 if r['arm']=='sentence-ab':
  b=idx['sentence-ba',r['id']];pairs.append(dict(id=r['id'],abRowSha256=rh(r),baRowSha256=rh(b),sameRequestAfterLabelSwap=True,samePrepared=r['prepared']==b['prepared'],normalizedChoiceAgrees=choices(r)==choices(b),sameEditedSource=r['editedSource']==b['editedSource']))
assert len(controls)==202 and len(pairs)==101
save(W/'manual-review-completion.json',dict(complete=True,actualResponses=404,controlRanges={'baseline':[[0,78],[78,101]],'balanced':[[0,78],[78,101]]},fullSourceContextRanges={'sentence-ab':[[0,40],[40,80],[80,101]],'sentence-ba':[[0,40],[40,80],[80,99],[99,101]]},failedUniqueRanges=[[0,20],[20,28]],changedBalancedRanges=[[0,16],[16,32]],scope='Assistant personally read actual raw output/source pairs, not an independent blind or model-judge review. Frozen evaluator unchanged.'))
shutil.copytree(W,P);D=P/'run'
save(D/'review.json',dict(resultsSha256=sha(D/'results.jsonl'),method='Every404 actual raw output and edited source personally read; all202 A/B cases with full source/context reviewed in explicit ranges, plus28 unique failed tuples and32 changed balanced-control pairs. Normalized operations and literal/semantic/required-repair judgments are separate from raw labels. Not independent blind acceptance.',reviews=reviews))
save(D/'control-parity.json',dict(records=controls,scope='202 newly executed controls match old actual requests and prepared inputs exactly; prior responses are comparison references, not newly counted inference.'))
save(D/'counterbalance-comparison.json',dict(records=pairs,scope='101 paired actual A/B requests change only option labels/ordering, corresponding example output labels and schema labels. Source/context/candidates and system remain identical. Normalized operation agreement is separate from correctness; optional letter normalization may change while both remain correct.'))
passed=[a for a in ['baseline','balanced','sentence-ab','sentence-ba'] if all(j['choiceConforms'] for j in reviews if j['variant']==a+'-selection')];assert not passed
save(D/'selection.json',dict(qualified=False,selected=None,screenPassedArms=passed,productionActivated=False,freshExecuted=False,formatterExecuted=False,resultsSha256=sha(D/'results.jsonl'),reviewSha256=sha(D/'review.json'),reason='Both sentence-choice arms fail:81/91 conforming versus94 balanced control; harmful19/5 versus4. Label counterbalance does not establish stable or correct semantic decisions. No formatter/fresh expansion or production adoption.'))
save(P/'path-mappings.json',dict(source='.workspace/rewrite-selector-sentence-choice-v1',published='benchmarks/runs/rewrite-selector-sentence-choice-v1',privateExcluded=['weights','runtime binaries','Go cleanup executable']))
shutil.copyfile(Path(__file__),H/'original-close.py.txt')
print(json.dumps(dict(groups={k:dict(v) for k,v in groups.items()},controlRawParity=sum(x['sameRawChoice'] for x in controls),counterbalanceAgreement=sum(x['normalizedChoiceAgrees'] for x in pairs))))
