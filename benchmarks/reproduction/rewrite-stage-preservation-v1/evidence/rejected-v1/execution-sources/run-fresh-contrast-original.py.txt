import os,json,subprocess,pathlib,hashlib
root=pathlib.Path(__file__).resolve().parents[3]
base=root/'.workspace/rewrite-stage-preservation'
out=base/'fresh-contrast-matrix'
config=json.loads((out/'protocol.json').read_text())
progress=[]
for arm in config['arms']:
 env=dict(os.environ)
 for key in list(env):
  if key.startswith('ARION_STAGE_'):del env[key]
 env.update(ARION_GGUF_RUNTIME='/Applications/Arion.app/Contents/Resources/llm',ARION_GGUF_MODEL='/Users/steven/Library/Application Support/Arion/models/qwen3.5-2b-q4-k-m-metal',ARION_STAGE_CASES=config['cases'],ARION_STAGE_OUT=str(out/(arm+'.jsonl')))
 overlay={'v3':config['v3Overlay'],'v1':str(base/'matrix-1/development-candidate/overlay.json'),'baseline':config['baselineOverlay']}[arm]
 command=['go','test','-overlay='+overlay,'./apps/desktop/infrastructure/llm/gguf','-run','^TestNativeStagePreservation$','-count=1','-timeout','15m']
 print('starting',arm,flush=True)
 with (out/(arm+'.log')).open('x') as log:result=subprocess.run(command,cwd=root,env=env,stdout=log,stderr=subprocess.STDOUT)
 progress.append(dict(arm=arm,command=command,exitCode=result.returncode))
 (out/'progress.json').write_text(json.dumps(progress,indent=2)+'\n')
 if result.returncode:raise SystemExit(result.returncode)
 print('completed',arm,flush=True)
(out/'complete.json').write_text(json.dumps({'status':'completed','arms':3,'quality':'Manual review required'},indent=2)+'\n')
