"""Materialize every failed source/context tuple and changed pair for personal review."""
import pathlib,json,hashlib,collections
R=pathlib.Path.cwd();W=R/'.workspace/rewrite-selector-contrast-demo-v1'
rows=list(map(json.loads,(W/'run/results.jsonl').read_text().splitlines()));assert len(rows)==404
E={x['id']:x for x in json.loads((R/'benchmarks/corpora/rewrite-selector-screen-v1/evaluator.json').read_text())}
def rh(x):return hashlib.sha256(json.dumps(x,ensure_ascii=False,sort_keys=True).encode()).hexdigest()
def record(r):return dict(arm=r['arm'],id=r['id'],rowSha256=rh(r),source=r['source'],context=r['context'],hypotheses=r['prepared']['hypotheses'],raw=r['response']['choices'][0]['message']['content'],selection=r['selection'],edited=r['editedSource'],wrongChoices={k:v for k,v in r['selection'].items() if v not in E[r['id']]['allowedChoices'][k]})
idx={(r['arm'],r['id']):r for r in rows};seen=set();failed=[];pairs=[]
for r in rows:
 x=record(r);key=rh({k:x[k] for k in ['source','context','hypotheses','selection','edited']})
 if x['wrongChoices'] and key not in seen:seen.add(key);failed.append(x)
 if r['arm'] in ['contrast-demo','contrast-demo-balanced']:
  ca='balanced' if r['arm']=='contrast-demo-balanced' else 'baseline';c=idx[(ca,r['id'])]
  if c['selection']!=r['selection']:pairs.append(dict(control=record(c),candidate=x))
out=dict(uniqueFailures=failed,changedPairs=pairs)
(W/'review-tuples.json').write_text(json.dumps(out,ensure_ascii=False,indent=2)+'\n')
print('uniqueFailures',len(failed),'changedPairs',len(pairs))
