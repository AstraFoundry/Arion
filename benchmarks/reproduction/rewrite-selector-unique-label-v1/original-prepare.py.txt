"""Freeze single-field unique-status omission before actual native inference."""
from pathlib import Path
import json,hashlib,shutil,subprocess
R=Path.cwd();H=R/'benchmarks/reproduction/rewrite-selector-unique-label-v1';W=R/'.workspace/rewrite-selector-unique-label-v1'
def read(p):return json.loads(p.read_text())
def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
def save(p,x):p.write_text(json.dumps(x,ensure_ascii=False,indent=2)+'\n')
shutil.copyfile(Path(__file__),H/'original-prepare.py.txt')
paths={a:'benchmarks/reproduction/rewrite-selector-screen-v1/templates/'+a+'/prompts.json' for a in ['baseline','balanced']}
save(H/'layout.json',dict(version='selector-unique-label-v1',systemAndExampleSources={a:dict(path=p,sha256=sha(R/p)) for a,p in paths.items()},transformation='Delete only status:unique from every example/current model-facing hypothesis. Keep ambiguous status, kind, every candidate spelling, source, complete OCR, scene, system and assistant demonstration decisions. Prepared producer output and JSON schema do not change.',ownership='Composite recipe with pinned original system/example prompt files and per-row complete message hashes. Reinsert missing unique status before kind to restore exact controls.'))
protocol=read(R/'benchmarks/reproduction/rewrite-selector-source-last-v1/protocol.json')
protocol.update(purpose='Test whether explicit unverified unique-status labels bias substitutions. Omit that field only; retain ambiguous labels and every other input.',arms=['baseline','no-unique-label','balanced','no-unique-label-balanced'],changes='Only model-facing hypothesis status:unique is removed in all examples/current input. Full context, spelling options, kind and ambiguity labels, source bytes, original source/replacement comparisons, trusted rules, assistant answers, actual producer, schema and native settings remain unchanged. The system unique-is-not-correct clause is retained; this is label exposure, not rule removal.')
save(H/'protocol.json',protocol)
f=read(R/'benchmarks/reproduction/rewrite-selector-screen-v1/freeze.json');f['files']=dict(read(R/'benchmarks/runs/rewrite-selector-screen-v1/archive.json')['files'])
for p in H.iterdir():
 if p.is_file() and p.name!='freeze.json':f['files'][str(p.relative_to(R))]=sha(p)
save(H/'freeze.json',f)
host={}
for key,cmd in [('os',['sw_vers']),('hardware',['sysctl','hw.model','hw.memsize','hw.physicalcpu','machdep.cpu.brand_string']),('power',['pmset','-g','custom']),('thermal',['pmset','-g','therm']),('gitRevision',['git','rev-parse','HEAD']),('dirty',['git','status','--porcelain']),('runtime',['.workspace/rewrite-small-models-v1/runtime/llama-b11734/llama-server','--version'])]:
 q=subprocess.run(cmd,text=True,capture_output=True);host[key]=dict(command=cmd,exitCode=q.returncode,stdout=q.stdout,stderr=q.stderr)
host.update(timer='time.monotonic',freezeSha256=sha(H/'freeze.json'),sampling='Actual1.88B Q4/b11734/Metal99/CPU4/ctx4096/parallel1/greedy42/nonthinking/cachetrue/128tokens/no warmup. RSS is not peak or actual16GB acceptance.',executorAdapter='Declared unique-label pipeline/freeze injected into unchanged four-arm actual-Go/Metal executor. Internal labels remain selector-screen-v1. New prompt identity owns only the metadata-omission recipe.')
save(W/'host-provenance.json',host);print('Prefrozen',len(f['files']),'files;404 planned choices,457 actualGo preparations')
